Constants & Variables


let name = "Alex"

let

Constant

var

Variable

Basic types


let name: String = "Alex"

let age: Int = 20

let price: Double = 19.99

let isStudent: Bool = true

String interpolation


let name = "Alex"
let age = 20

print("My name is \(name)")
print("I am \(age) years old")

Swift can insert values directly into strings.

Expressions


let a = 10
let b = 3

print(a + b)
print(a - b)
print(a * b)
print(a / b)

Making decisions


let age = 20

if age >= 18 {
    print("Adult")
} else {
    print("Minor")
}

Repeating work


for number in 1...5 {
    print(number)
}

Loops let us process multiple values.

Functions


func greet(name: String) {
    print("Hello, \(name)!")
}

greet(name: "Alex")

Function = reusable piece of behavior

Functions return values


func add(_ a: Int, _ b: Int) -> Int {
    return a + b
}

let result = add(10, 20)

print(result)
